// eval/acceptance/refunds.go
func (h *Hooks) InstantCredit(
  ctx context.Context,
) (eval.Result, error) {
  // $50.00 subtotal + $4.50 tax
  out, err := h.refund(ctx, 5000, 450)
  if err != nil {
    return eval.Result{}, err
  }
  return eval.Result{Checks: []eval.Check{
    check("50_subtotal", out.Status == "credit"),
    check("cents", out.Cents == 5450),
  }}, nil
}
